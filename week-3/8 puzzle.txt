import heapq

goal = (1,2,3,4,5,6,7,8,0)

def h(s):
    return sum(abs(i//3 - goal.index(v)//3) + abs(i%3 - goal.index(v)%3)
               for i,v in enumerate(s) if v != 0)

def neigh(s):
    i = s.index(0)
    x,y = divmod(i,3)
    for dx,dy in [(-1,0),(1,0),(0,-1),(0,1)]:
        nx,ny = x+dx,y+dy
        if 0<=nx<3 and 0<=ny<3:
            ni = nx*3+ny
            t = list(s)
            t[i],t[ni] = t[ni],t[i]
            yield tuple(t)

def astar(start):
    pq = [(h(start),0,start)]
    parent = {}
    g = {start:0}

    while pq:
        _,cost,s = heapq.heappop(pq)
        if s == goal:
            path = [s]
            while s in parent:
                s = parent[s]
                path.append(s)
            return path[::-1]
        for n in neigh(s):
            ng = g[s] + 1
            if n not in g or ng < g[n]:
                g[n] = ng
                heapq.heappush(pq, (ng+h(n), ng, n))
                parent[n] = s

start = (1,2,4,6,5,0,7,8,3)
solution = astar(start)
for step, state in enumerate(solution):
    print(f"\nStep {step}")
    print("-------------")
    for i in range(0,9,3):
        # Print each row as numbers, replace 0 with blank
        row = [' ' if x==0 else x for x in state[i:i+3]]
        print("|", row[0], row[1], row[2], "|")
    print("-------------")

print("\nTotal Moves =", len(solution)-1)
