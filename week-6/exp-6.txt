# Step 1: Import Libraries
from pgmpy.models import DiscreteBayesianNetwork
from pgmpy.inference import VariableElimination
import pandas as pd

# Step 2: Define Dataset
data = pd.DataFrame({
    'Rain': ['No', 'No', 'Yes', 'Yes', 'No', 'Yes', 'Yes', 'No'],
    'TrafficJam': ['Yes', 'No', 'Yes', 'No', 'Yes', 'Yes', 'No', 'No'],
    'ArriveLate': ['Yes', 'No', 'Yes', 'No', 'No', 'Yes', 'Yes', 'No']
})

print("Dataset:\n", data)

# Step 3: Define Bayesian Network Structure (DAG)
model = DiscreteBayesianNetwork([
    ('Rain', 'TrafficJam'),
    ('TrafficJam', 'ArriveLate')
])

# Step 4: Parameter Learning using MLE
model.fit(data)

# Step 5: Display Conditional Probability Distributions (CPDs)
print("\nLearned CPDs:")
for cpd in model.get_cpds():
    print(cpd)
    print()

# Step 6: Initialize Inference Engine
inference = VariableElimination(model)

# Step 7: Perform Query
# P(ArriveLate | Rain = Yes)
query_result = inference.query(
    variables=['ArriveLate'],
    evidence={'Rain': 'Yes'}
)

# Step 8: Display Result
print("\nInference Result:")
print(query_result)
