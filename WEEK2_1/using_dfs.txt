def water_jug_dfs_3jugs(capA, capB, capC, target):
    visited = set()
    path = []

    def dfs(a, b, c):
        if (a, b, c) in visited:
            return False

        visited.add((a, b, c))
        path.append((a, b, c))

        if a == target or b == target or c == target:
            return True

        next_states = [
            (capA, b, c),
            (a, capB, c),
            (a, b, capC),
            (0, b, c),
            (a, 0, c),
            (a, b, 0)
        ]

        t = min(a, capB - b)
        next_states.append((a - t, b + t, c))

        t = min(a, capC - c)
        next_states.append((a - t, b, c + t))

        t = min(b, capA - a)
        next_states.append((a + t, b - t, c))

        t = min(b, capC - c)
        next_states.append((a, b - t, c + t))

        t = min(c, capA - a)
        next_states.append((a + t, b, c - t))

        t = min(c, capB - b)
        next_states.append((a, b + t, c - t))

        for state in next_states:
            if dfs(state[0], state[1], state[2]):
                return True

        path.pop()
        return False

    if dfs(0, 0, 0):
        return path
    else:
        return None
result = water_jug_dfs_3jugs(8, 5, 3, 4)

if result:
    for step in result:
        print(step)
else:
    print("No solution")
